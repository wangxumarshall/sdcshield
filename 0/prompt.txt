#!/bin/bash
# 全核(0-607)并发跑 ./testwang,每核一个实例,持续 1 小时
cd "$(dirname "$0")"
mkdir -p logs
end=$((SECONDS + 3600))
round=0
while [ $SECONDS -lt $end ]; do
    round=$((round + 1))
    echo "[round $round] $(date +%T)"
    for c in $(seq 0 607); do
        taskset -c "$c" ./testwang >> "logs/tw_$c.log" 2>&1 &
    done
    wait
done
echo "finished: $round rounds"
